**Tech Stack:** Go, net/smtp, net/http。

package message

import "context"

// Message 表示一条待发送的消息。
type Message struct {
	To       string            // 收件人：邮箱地址 / 手机号
	Subject  string            // 主题（email 用）
	Body     string            // 正文内容
	Template string            // 模板 ID（可选，provider 支持时使用）
	Data     map[string]string // 模板变量（可选）
}

// Provider 消息发送 provider 接口。
type Provider interface {
	// Channel 返回渠道类型："email" / "sms"
	Channel() string
	// Name 返回 provider 名称，如 "smtp", "mailgun", "aliyun"
	Name() string
	// Send 发送消息。返回 nil 表示成功。
	Send(ctx context.Context, msg *Message) error
}

// message/message.go
package message

import (
	"context"
	"fmt"
	"log/slog"
	"sync"
)

// Messenger 消息发送器，支持多渠道多 provider fallback。
type Messenger struct {
	mu        sync.RWMutex
	providers map[string][]Provider // channel → providers（按优先级排序）
}

// NewMessenger 创建 Messenger。providers 按 channel 分组，同一 channel 内按传入顺序优先级递减。
func NewMessenger(providers []Provider) (*Messenger, error) {
	m := &Messenger{
		providers: make(map[string][]Provider),
	}
	for _, p := range providers {
		ch := p.Channel()
		m.providers[ch] = append(m.providers[ch], p)
	}
	return m, nil
}

// Send 通过指定渠道发送消息。自动选择 provider，失败时 fallback 到下一个。
func (m *Messenger) Send(ctx context.Context, channel string, msg *Message) error {
	if err := ctx.Err(); err != nil {
		return fmt.Errorf("message send cancelled: %w", err)
	}

	m.mu.RLock()
	providers := m.providers[channel]
	m.mu.RUnlock()

	if len(providers) == 0 {
		return fmt.Errorf("no provider for channel %q", channel)
	}

	var lastErr error
	for _, p := range providers {
		if err := ctx.Err(); err != nil {
			return fmt.Errorf("message send cancelled: %w", err)
		}
		if err := p.Send(ctx, msg); err != nil {
			slog.Warn("provider send failed, trying next", "channel", channel, "provider", p.Name(), "error", err)
			lastErr = err
			continue
		}
		return nil
	}
	return fmt.Errorf("all providers failed for channel %q: %w", channel, lastErr)
}
```

// message/email/smtp.go
package email

import (
	"context"
	"fmt"
	"net/smtp"
	"strings"

	"go-common/message"
)

// SMTPConfig SMTP provider 配置。
type SMTPConfig struct {
	Host     string
	Port     int
	Username string
	Password string
	From     string
}

// SMTPProvider 通过 SMTP 发送邮件。
type SMTPProvider struct {
	config SMTPConfig
}

// NewSMTPProvider 创建 SMTP provider。
func NewSMTPProvider(config SMTPConfig) *SMTPProvider {
	return &SMTPProvider{config: config}
}

func (p *SMTPProvider) Channel() string { return "email" }
func (p *SMTPProvider) Name() string    { return "smtp" }

func (p *SMTPProvider) Send(_ context.Context, msg *message.Message) error {
	if msg.To == "" {
		return fmt.Errorf("smtp: recipient is empty")
	}

	addr := fmt.Sprintf("%s:%d", p.config.Host, p.config.Port)
	subject := msg.Subject
	if subject == "" {
		subject = "Notification"
	}

	// 构建邮件内容
	var body strings.Builder
	body.WriteString("From: " + p.config.From + "\r\n")
	body.WriteString("To: " + msg.To + "\r\n")
	body.WriteString("Subject: " + subject + "\r\n")
	body.WriteString("MIME-Version: 1.0\r\n")
	body.WriteString("Content-Type: text/plain; charset=\"utf-8\"\r\n")
	body.WriteString("\r\n")
	body.WriteString(msg.Body)

	auth := smtp.PlainAuth("", p.config.Username, p.config.Password, p.config.Host)
	return smtp.SendMail(addr, auth, p.config.From, []string{msg.To}, []byte(body.String()))
}
```

// message/sms/aliyun.go
package sms

import (
	"bytes"
	"context"
	"crypto/hmac"
	"crypto/sha1"
	"encoding/base64"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"net/url"
	"sort"
	"strings"
	"time"

	"go-common/message"
)

// AliyunConfig 阿里云短信配置。
type AliyunConfig struct {
	AccessKeyID     string
	AccessKeySecret string
	SignName        string
	Templates       map[string]string // purpose → template_code
}

// AliyunProvider 通过阿里云 SMS API 发送短信。
type AliyunProvider struct {
	config AliyunConfig
	client *http.Client
}

// NewAliyunProvider 创建阿里云 SMS provider。
func NewAliyunProvider(config AliyunConfig) *AliyunProvider {
	return &AliyunProvider{
		config: config,
		client: &http.Client{Timeout: 10 * time.Second},
	}
}

func (p *AliyunProvider) Channel() string { return "sms" }
func (p *AliyunProvider) Name() string    { return "aliyun" }

func (p *AliyunProvider) Send(_ context.Context, msg *message.Message) error {
	if msg.To == "" {
		return fmt.Errorf("aliyun sms: phone number is empty")
	}

	templateCode := p.resolveTemplate(msg)
	if templateCode == "" {
		return fmt.Errorf("aliyun sms: no template code for message")
	}

	params := map[string]string{
		"AccessKeyId":      p.config.AccessKeyID,
		"Action":           "SendSms",
		"Format":           "JSON",
		"PhoneNumbers":     msg.To,
		"SignName":         p.config.SignName,
		"SignatureMethod":  "HMAC-SHA1",
		"SignatureNonce":   fmt.Sprintf("%d", time.Now().UnixNano()),
		"SignatureVersion": "1.0",
		"TemplateCode":     templateCode,
		"Timestamp":        time.Now().UTC().Format("2006-01-02T15:04:05Z"),
		"Version":          "2017-05-25",
	}

	if msg.Data != nil {
		templateParam, _ := json.Marshal(msg.Data)
		params["TemplateParam"] = string(templateParam)
	}

	// 排序并构造签名字符串
	sortedKeys := make([]string, 0, len(params))
	for k := range params {
		sortedKeys = append(sortedKeys, k)
	}
	sort.Strings(sortedKeys)

	var queryParts []string
	for _, k := range sortedKeys {
		queryParts = append(queryParts, specialEncode(k)+"="+specialEncode(params[k]))
	}
	canonicalQuery := strings.Join(queryParts, "&")

	stringToSign := "GET&" + specialEncode("/") + "&" + specialEncode(canonicalQuery)

	mac := hmac.New(sha1.New, []byte(p.config.AccessKeySecret+"&"))
	mac.Write([]byte(stringToSign))
	signature := base64.StdEncoding.EncodeToString(mac.Sum(nil))

	reqURL := "https://dysmsapi.aliyuncs.com/?" + canonicalQuery + "&Signature=" + specialEncode(signature)

	resp, err := p.client.Get(reqURL)
	if err != nil {
		return fmt.Errorf("aliyun sms request: %w", err)
	}
	defer resp.Body.Close()

	body, _ := io.ReadAll(resp.Body)
	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("aliyun sms: status %d, body: %s", resp.StatusCode, string(body))
	}

	var result struct {
		Code string `json:"Code"`
	}
	if err := json.Unmarshal(body, &result); err != nil {
		return fmt.Errorf("aliyun sms parse response: %w", err)
	}
	if result.Code != "OK" {
		return fmt.Errorf("aliyun sms: code=%s, body: %s", result.Code, string(body))
	}
	return nil
}

func (p *AliyunProvider) resolveTemplate(msg *message.Message) string {
	if msg.Template != "" {
		return msg.Template
	}
	return ""
}

func specialEncode(s string) string {
	return url.QueryEscape(s)
}
```

// message/email/mailgun.go
package email

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"net/url"
	"time"

	"go-common/message"
)

// MailgunConfig Mailgun provider 配置。
type MailgunConfig struct {
	Domain string
	APIKey string
	From   string
}

// MailgunProvider 通过 Mailgun HTTP API 发送邮件。
type MailgunProvider struct {
	config MailgunConfig
	client *http.Client
}

// NewMailgunProvider 创建 Mailgun provider。
func NewMailgunProvider(config MailgunConfig) *MailgunProvider {
	return &MailgunProvider{
		config: config,
		client: &http.Client{Timeout: 10 * time.Second},
	}
}

func (p *MailgunProvider) Channel() string { return "email" }
func (p *MailgunProvider) Name() string    { return "mailgun" }

func (p *MailgunProvider) Send(_ context.Context, msg *message.Message) error {
	if msg.To == "" {
		return fmt.Errorf("mailgun: recipient is empty")
	}

	apiURL := fmt.Sprintf("https://api.mailgun.net/v3/%s/messages", p.config.Domain)
	data := url.Values{
		"from":    {p.config.From},
		"to":      {msg.To},
		"subject": {msg.Subject},
		"text":    {msg.Body},
	}

	req, err := http.NewRequestWithContext(context.Background(), "POST", apiURL, bytes.NewBufferString(data.Encode()))
	if err != nil {
		return fmt.Errorf("mailgun create request: %w", err)
	}
	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	req.SetBasicAuth("api", p.config.APIKey)

	resp, err := p.client.Do(req)
	if err != nil {
		return fmt.Errorf("mailgun send: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode >= 300 {
		body, _ := io.ReadAll(resp.Body)
		return fmt.Errorf("mailgun: status %d, body: %s", resp.StatusCode, string(body))
	}

	// 解析响应确认成功
	var result map[string]any
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		// 响应不是 JSON，但状态码正常，视为成功
		return nil
	}
	return nil
}
```

import (

func TestMessenger_Integration_emailFallback(t *testing.T) {

	m, err := NewMessenger([]Provider{smtp, mailgun})

	msg := &Message{To: "user@test.com", Subject: "Verify", Body: "Your code is 123456"}
	err = m.Send(context.Background(), "email", msg)
	require.NoError(t, err)

	assert.Nil(t, smtp.sent)   // smtp 失败
	assert.Equal(t, msg, mailgun.sent) // mailgun 成功
}

func TestMessenger_Integration_mixedChannels(t *testing.T) {
	emailP := &mockProvider{channel: "email", name: "smtp"}
	smsP := &mockProvider{channel: "sms", name: "aliyun"}

	m, err := NewMessenger([]Provider{emailP, smsP})
	require.NoError(t, err)

	// 发 SMS
	smsMsg := &Message{To: "13800138000", Body: "123456"}
	err = m.Send(context.Background(), "sms", smsMsg)
	require.NoError(t, err)
	assert.Equal(t, smsMsg, smsP.sent)
	assert.Nil(t, emailP.sent)

	// 重置
	smsP.sent = nil

	// 发 Email
	emailMsg := &Message{To: "user@test.com", Subject: "Hi", Body: "Hello"}
	err = m.Send(context.Background(), "email", emailMsg)
	require.NoError(t, err)
	assert.Equal(t, emailMsg, emailP.sent)
	assert.Nil(t, smsP.sent)
}
