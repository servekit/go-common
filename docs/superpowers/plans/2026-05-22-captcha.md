**Goal:** 重构 captcha 模块为 Config struct 驱动，内部封装 RedisStore/RateLimiter/CodeGenerator，TTL 从限流规则自动推导。

**Architecture:** `Config` struct 聚合所有配置（可从配置文件加载），`New(cfg, opts...)` 内部创建所有依赖。`PurposeConfig` 按 purpose 聚合限流规则、验证码格式、渠道配置。TTL = 该 purpose 最短限流窗口。

**Tech Stack:** Go, go-redis/redis/v9, go-common/message, go-common/ratelimit

package captcha

import (
	"context"
	"fmt"
	"time"

	"github.com/google/uuid"
	"github.com/redis/go-redis/v9"

	"code.byteflowing.com/base/go-common/ratelimit"
)

// Channel represents the delivery channel for verification codes.
type Channel string

const (
	ChannelEmail Channel = "email"
	ChannelSMS   Channel = "sms"
)

// RedisConfig holds Redis connection info (typically from config file).
type RedisConfig struct {
	Addr     string
	Password string
	DB       int
}

// Config holds all captcha service configuration.
// Can be loaded from config file (YAML/JSON etc).
type Config struct {
	Prefix      string                   // Redis key prefix, default "captcha"
	Redis       RedisConfig              // Redis connection info, required if no WithRedisClient option
	MaxAttempts int                       // max verify attempts per captcha, default 3
	GlobalRules []ratelimit.Rule          // global rate limits applied across all purposes
	Purposes    map[string]PurposeConfig  // purpose → config
}

// PurposeConfig groups all settings for a single purpose.
type PurposeConfig struct {
	CodeFormat CodeFormat        // verification code format, default FormatDigit6
	RateRules  []ratelimit.Rule  // rate limit rules, shortest window = TTL
	Channels   []ChannelConfig   // supported delivery channels
}

// ChannelConfig describes a delivery channel for a purpose.
type ChannelConfig struct {
	Channel Channel
	Sender  ChannelSender
}

// Option configures a Captcha instance.
type Option func(*Captcha)

// WithRedisClient provides an existing Redis client instance.
// Overrides Config.Redis — no new connection is created.
func WithRedisClient(client *redis.Client) Option {
	return func(c *Captcha) { c.client = client }
}

// Captcha is the verification code service.
type Captcha struct {
	client      *redis.Client
	store       Store
	limiter     ratelimit.Limiter
	codeGen     *CodeGenerator
	channels    map[string]map[Channel]ChannelSender // purpose → channel → sender
	purposes    map[string]PurposeConfig
	maxAttempts int
	prefix      string
}

const defaultPrefix = "captcha"
const defaultTTL = 5 * time.Minute

// New creates a Captcha service from config + optional overrides.
func New(cfg Config, opts ...Option) (*Captcha, error) {
	c := &Captcha{
		prefix:      cfg.Prefix,
		maxAttempts: cfg.MaxAttempts,
		purposes:    cfg.Purposes,
		channels:    make(map[string]map[Channel]ChannelSender),
	}

	if c.prefix == "" {
		c.prefix = defaultPrefix
	}
	if c.maxAttempts <= 0 {
		c.maxAttempts = 3
	}

	// Apply options (may set client via WithRedisClient).
	for _, opt := range opts {
		opt(c)
	}

	// If no client from options, create one from Config.Redis.
	if c.client == nil {
		if cfg.Redis.Addr == "" {
			return nil, fmt.Errorf("captcha: redis addr is required when no client is provided")
		}
		c.client = redis.NewClient(&redis.Options{
			Addr:     cfg.Redis.Addr,
			Password: cfg.Redis.Password,
			DB:       cfg.Redis.DB,
		})
	}

	// Build rate limiter from all purpose rules + global rules.
	rateRules := make(map[string][]ratelimit.Rule)
	for purpose, pc := range cfg.Purposes {
		if len(pc.RateRules) > 0 {
			rateRules[purpose] = pc.RateRules
		}
	}
	c.limiter = ratelimit.NewRedisLimiter(c.client, ratelimit.Config{
		Prefix: c.prefix + ":rate",
		Global: cfg.GlobalRules,
		Rules:  rateRules,
	})

	// Build code generator from purpose formats.
	formats := make(map[string]CodeFormat)
	for purpose, pc := range cfg.Purposes {
		formats[purpose] = pc.CodeFormat
	}
	c.codeGen = NewCodeGenerator(formats)

	// Build store.
	c.store = NewRedisStore(c.client, c.prefix)

	// Build channels map: purpose → channel → sender.
	for purpose, pc := range cfg.Purposes {
		chMap := make(map[Channel]ChannelSender)
		for _, cc := range pc.Channels {
			chMap[cc.Channel] = cc.Sender
		}
		c.channels[purpose] = chMap
	}

	return c, nil
}

// ttlForPurpose returns the TTL for a purpose derived from its rate rules.
func (c *Captcha) ttlForPurpose(purpose string) time.Duration {
	pc, ok := c.purposes[purpose]
	if !ok || len(pc.RateRules) == 0 {
		return defaultTTL
	}
	min := pc.RateRules[0].Window
	for _, r := range pc.RateRules[1:] {
		if r.Window < min {
			min = r.Window
		}
	}
	return min
}

// Send generates, stores, and delivers a verification code.
func (c *Captcha) Send(ctx context.Context, target string, channel Channel, purpose string) (string, error) {
	if target == "" {
		return "", fmt.Errorf("captcha: target is empty")
	}
	if purpose == "" {
		return "", fmt.Errorf("captcha: purpose is empty")
	}

	pc, ok := c.purposes[purpose]
	if !ok {
		return "", fmt.Errorf("captcha: purpose %q not configured", purpose)
	}

	ch, ok := c.channels[purpose][channel]
	if !ok {
		return "", fmt.Errorf("captcha: channel %q not configured for purpose %q", channel, purpose)
	}

	// Rate limit check.
	if c.limiter != nil {
		allowed, err := c.limiter.Allow(purpose, target)
		if err != nil {
			return "", fmt.Errorf("check rate limit: %w", err)
		}
		if !allowed {
			return "", fmt.Errorf("rate limit exceeded for %s", target)
		}
	}

	// Generate code.
	code, err := c.codeGen.Generate(purpose)
	if err != nil {
		return "", fmt.Errorf("generate code: %w", err)
	}

	ttl := c.ttlForPurpose(purpose)
	captchaID := uuid.New().String()

	record := &Record{
		CaptchaID:   captchaID,
		Code:        code,
		Target:      target,
		Purpose:     purpose,
		MaxAttempts: c.maxAttempts,
	}

	if err := c.store.Set(ctx, purpose, target, record, ttl); err != nil {
		return "", fmt.Errorf("store code: %w", err)
	}

	if err := ch.SendCode(ctx, target, code, purpose); err != nil {
		return "", fmt.Errorf("send code: %w", err)
	}

	return captchaID, nil
}

// Verify checks a verification code by target and purpose.
func (c *Captcha) Verify(ctx context.Context, target, code, purpose string) (*VerifyResult, error) {
	return c.store.Verify(ctx, purpose, target, code)
}

func TestCaptcha_ttlForPurpose(t *testing.T) {
	tests := []struct {
		name     string
		config   Config
		purpose  string
		expected time.Duration
	}{
		{
			name: "shortest window from rate rules",
			config: Config{
				Purposes: map[string]PurposeConfig{
					"register": {
						RateRules: []ratelimit.Rule{
							{Window: 5 * time.Minute, Max: 10},
							{Window: time.Hour, Max: 20},
						},
					},
				},
			},
			purpose:  "register",
			expected: 5 * time.Minute,
		},
		{
			name: "no rate rules uses default",
			config: Config{
				Purposes: map[string]PurposeConfig{
					"register": {},
				},
			},
			purpose:  "register",
			expected: defaultTTL,
		},
		{
			name: "unknown purpose uses default",
			config: Config{
				Purposes: map[string]PurposeConfig{},
			},
			purpose:  "unknown",
			expected: defaultTTL,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			c, err := New(tt.config, WithRedisClient(testRedisClient(t)))
			require.NoError(t, err)
			got := c.ttlForPurpose(tt.purpose)
			require.Equal(t, tt.expected, got)
		})
	}
}
