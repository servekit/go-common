**Goal:** Implement `configx`, a general-purpose config loading library in go-common that wraps viper with a single `Load()` function.

**Architecture:** Four files — `errors.go` defines xerr error codes; `options.go` defines `LoadOption` and `With*` functions; `configx.go` implements `Load()` with file resolution (pflag → env → filesystem search), viper setup, decode hooks, and tagless struct matching via `MatchName`; `configx_test.go` covers all behaviors.

**Tech Stack:** Go 1.25, viper, pflag, mapstructure, xerr

```go
package configx

import "github.com/mitchellh/mapstructure"

// LoadOption configures the config loading behavior.
type LoadOption func(*loader)

// WithServiceName sets the service name.
// This enables <NAME>_CONFIG env var lookup and adds /etc/<name> to search paths.
func WithServiceName(name string) LoadOption {
	return func(l *loader) {
		l.serviceName = name
	}
}

// WithEnvPrefix sets the viper env prefix for AutomaticEnv (e.g. "MY_SERVICE").
func WithEnvPrefix(prefix string) LoadOption {
	return func(l *loader) {
		l.envPrefix = prefix
	}
}

// WithConfigName sets the config file name without extension (default: "config").
func WithConfigName(name string) LoadOption {
	return func(l *loader) {
		l.configName = name
	}
}

// WithConfigPaths sets the directories to search for config files.
func WithConfigPaths(paths ...string) LoadOption {
	return func(l *loader) {
		l.configPaths = paths
	}
}

// WithDefaults sets viper default values.
func WithDefaults(m map[string]any) LoadOption {
	return func(l *loader) {
		l.defaults = m
	}
}

// WithDecodeHooks appends custom mapstructure decode hooks after the built-in ones.
func WithDecodeHooks(hooks ...mapstructure.DecodeHookFunc) LoadOption {
	return func(l *loader) {
		l.decodeHooks = append(l.decodeHooks, hooks...)
	}
}
```

```go
// Package configx provides a general-purpose configuration loading library.
// It wraps viper to offer a single Load() function that handles config file
// resolution (pflag → env → filesystem search), environment variable binding,
// structured decoding with built-in hooks, and tagless struct matching.
package configx

import (
	"fmt"
	"os"
	"strings"
	"sync"
	"time"

	"github.com/mitchellh/mapstructure"
	"github.com/spf13/pflag"
	"github.com/spf13/viper"
)

var parseOnce sync.Once

// File extensions supported by viper, in search priority order.
var supportedExts = []string{"yaml", "yml", "json", "toml", "hcl", "ini", "properties"}

// loader holds all configuration for the loading process.
type loader struct {
	serviceName string
	envPrefix   string
	configName  string
	configPaths []string
	defaults    map[string]any
	decodeHooks []mapstructure.DecodeHookFunc
}

func newLoader(opts []LoadOption) *loader {
	l := &loader{
		configName:  "config",
		configPaths: []string{"."},
	}
	for _, opt := range opts {
		opt(l)
	}
	if l.serviceName != "" {
		l.configPaths = append(l.configPaths, fmt.Sprintf("/etc/%s", l.serviceName))
	}
	return l
}

// resolveConfigPath determines the config file path in order:
//  1. -config pflag
//  2. <SERVICE_NAME>_CONFIG env var (if WithServiceName is set)
//  3. Filesystem search in configPaths
func (l *loader) resolveConfigPath() (string, error) {
	// 1. pflag
	cfgFile, _ := pflag.CommandLine.GetString("config")
	if cfgFile != "" {
		return cfgFile, nil
	}

	// 2. env var
	if l.serviceName != "" {
		envKey := strings.ToUpper(l.serviceName) + "_CONFIG"
		if env := os.Getenv(envKey); env != "" {
			return env, nil
		}
	}

	// 3. filesystem search
	for _, dir := range l.configPaths {
		for _, ext := range supportedExts {
			path := fmt.Sprintf("%s/%s.%s", dir, l.configName, ext)
			if _, err := os.Stat(path); err == nil {
				return path, nil
			}
		}
	}

	return "", ErrConfigNotFound.New()
}

// buildDecodeHook composes built-in hooks with user-provided ones.
func (l *loader) buildDecodeHook() mapstructure.DecodeHookFunc {
	hooks := []mapstructure.DecodeHookFunc{
		mapstructure.StringToTimeDurationHookFunc(),
		mapstructure.StringToSliceHookFunc(","),
		mapstructure.StringToTimeHookFunc(time.RFC3339),
	}
	hooks = append(hooks, l.decodeHooks...)
	return mapstructure.ComposeDecodeHookFunc(hooks...)
}

// Load reads configuration into target.
// Config file is resolved in order:
//  1. -config pflag (e.g. -config /etc/app/config.yaml)
//  2. <SERVICE_NAME>_CONFIG env var (if WithServiceName is set)
//  3. Filesystem search: config.yaml/config.json/... in configured paths
//
// Environment variables override file values.
// Built-in decode hooks handle time.Duration, []string, and time.Time.
// Struct fields do not need mapstructure tags — snake_case config keys
// are automatically matched to CamelCase field names.
func Load(target any, opts ...LoadOption) error {
	l := newLoader(opts)

	// Register -config flag (idempotent).
	if pflag.CommandLine.Lookup("config") == nil {
		pflag.String("config", "", "path to config file")
	}
	parseOnce.Do(func() { pflag.Parse() })

	// Resolve config file path.
	cfgPath, err := l.resolveConfigPath()
	if err != nil {
		return err
	}

	// Setup viper.
	v := viper.New()
	v.SetConfigFile(cfgPath)
	v.SetEnvKeyReplacer(strings.NewReplacer(".", "_"))
	v.AutomaticEnv()
	if l.envPrefix != "" {
		v.SetEnvPrefix(l.envPrefix)
	}

	// Set defaults.
	for key, val := range l.defaults {
		v.SetDefault(key, val)
	}

	// Read config file.
	if err := v.ReadInConfig(); err != nil {
		return ErrReadConfig.Wrap(err)
	}

	// Unmarshal with decode hooks and tagless matching.
	if err := v.Unmarshal(target, func(dc *mapstructure.DecoderConfig) {
		dc.DecodeHook = l.buildDecodeHook()
		dc.MatchName = func(mapKey, fieldName string) bool {
			return strings.EqualFold(
				strings.ReplaceAll(mapKey, "_", ""),
				fieldName,
			)
		}
	}); err != nil {
		return ErrUnmarshal.Wrap(err)
	}

	return nil
}
```
