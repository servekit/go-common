> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wrap `robfig/cron/v3` with Config + New() pattern, integrating slog logging, panic recovery, overlap policies, and an extension mechanism via Option.

**Architecture:** Follow the same pattern as redisx/dbx — `Config` struct with `mapstructure` tags, `New(cfg *Config, opts ...Option)` constructor returning `*cron.Cron`. Add a `slogLogger` adapter implementing `cron.Logger`. A lightweight `cronOptions` struct collects extra `cron.Option` values from `WithCronOption()`.

**Tech Stack:** Go 1.25, `robfig/cron/v3`, `log/slog`, `testify/assert`

```go
// Package cronx provides cron scheduler initialization with standardized configuration.
package cronx

import (
	"log/slog"
)

// slogLogger adapts log/slog to the cron.Logger interface.
type slogLogger struct {
	level string
}

func newSlogLogger(level string) *slogLogger {
	return &slogLogger{level: level}
}

func (l *slogLogger) Info(msg string, keysAndValues ...any) {
	if l.level != "info" {
		return
	}
	slog.Info(msg, keysAndValues...)
}

func (l *slogLogger) Error(err error, msg string, keysAndValues ...any) {
	if l.level == "silent" {
		return
	}
	keysAndValues = append(keysAndValues, "error", err)
	slog.Error(msg, keysAndValues...)
}
```

```go
package cronx

import (
	"fmt"
	"time"

	"github.com/robfig/cron/v3"
)

// Config holds cron scheduler parameters.
type Config struct {
	Timezone      string `mapstructure:"timezone"`       // IANA timezone, empty = Local
	WithSeconds   bool   `mapstructure:"with_seconds"`   // enable second-level precision
	LogLevel      string `mapstructure:"log_level"`      // silent | error | info (default: info)
	OverlapPolicy string `mapstructure:"overlap_policy"` // skip | delay | empty
}

// Option configures additional cron behavior.
type Option func(*cronOptions)

type cronOptions struct {
	extraOpts []cron.Option
}

// WithCronOption passes through a raw cron.Option for advanced customization.
func WithCronOption(opt cron.Option) Option {
	return func(o *cronOptions) {
		o.extraOpts = append(o.extraOpts, opt)
	}
}

// New creates a *cron.Cron with the given config and options.
func New(cfg *Config, opts ...Option) (*cron.Cron, error) {
	// Parse timezone
	loc := time.Local
	if cfg.Timezone != "" {
		var err error
		loc, err = time.LoadLocation(cfg.Timezone)
		if err != nil {
			return nil, fmt.Errorf("cronx: invalid timezone %q: %w", cfg.Timezone, err)
		}
	}

	// Collect extra options
	co := &cronOptions{}
	for _, opt := range opts {
		opt(co)
	}

	// Build slog logger
	logLevel := cfg.LogLevel
	if logLevel == "" {
		logLevel = "info"
	}
	logger := newSlogLogger(logLevel)

	// Build cron options
	cronOpts := []cron.Option{
		cron.WithLocation(loc),
		cron.WithLogger(logger),
	}

	if cfg.WithSeconds {
		cronOpts = append(cronOpts, cron.WithSeconds())
	}

	// Build job wrappers — always recover panics
	wrappers := []cron.JobWrapper{
		cron.Recover(logger),
	}

	switch cfg.OverlapPolicy {
	case "skip":
		wrappers = append(wrappers, cron.SkipIfStillRunning(logger))
	case "delay":
		wrappers = append(wrappers, cron.DelayIfStillRunning(logger))
	}

	cronOpts = append(cronOpts, cron.WithChain(wrappers...))

	// Append caller-provided options
	cronOpts = append(cronOpts, co.extraOpts...)

	return cron.New(cronOpts...), nil
}
```
